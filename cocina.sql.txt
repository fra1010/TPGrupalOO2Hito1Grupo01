create schema epicentro_gourmet;

use epicentro_gourmet;

create table festival(
id_festival int primary key auto_increment,
nombre varchar(30) not null,
temporada varchar(30) not null,
fecha_inicio date not null,
fecha_fin date not null
);
create table costo(
id_costo int primary key ,
costo_superficie int not null,
costo_montaje int not null,
costo_electricidad int not null ,
sueldo_base int not null,
 FOREIGN KEY (id_costo) REFERENCES festival(id_festival)
);
CREATE TABLE unidad_venta (
    id_unidad_venta INT PRIMARY KEY AUTO_INCREMENT ,
    nombre VARCHAR(50) NOT NULL,
    superficie double NOT NULL,
    codigo VARCHAR(50) UNIQUE NOT NULL,
    responsable_id INT  NULL unique,
   -- FOREIGN KEY (responsable_id) REFERENCES empleado(id_empleado),
    id_festival int ,
	FOREIGN KEY (id_festival) REFERENCES festival(id_festival)
    
);
create table empleado(
id_empleado int primary key auto_increment,
nombre varchar(30) not null,
apellido varchar(30) not null,
dni long not null,
fecha_nacimiento date not null,
ingreso date not null,
unidad_venta_id int NULL
-- FOREIGN KEY (unidad_venta_id) REFERENCES unidad_venta(id_unidad_venta)
);

create table cocinero(
id_empleado INT PRIMARY KEY not null,
especialidad varchar(30) not null,
 FOREIGN KEY (id_empleado) REFERENCES empleado(id_empleado)
);
create table cajero(
id_empleado INT PRIMARY KEY not null,
turno varchar(30) not null,
 FOREIGN KEY (id_empleado) REFERENCES empleado(id_empleado)
);





create table food_truck(
id_unidad_venta INT PRIMARY KEY not null,
patente varchar(40) NOT NULL,
conexion bit(1) not null,
 FOREIGN KEY (id_unidad_venta) REFERENCES unidad_venta (id_unidad_venta)
);
create table puesto_desarmable(
id_unidad_venta INT PRIMARY KEY not null,
cantidad_carpa int NOT NULL,
tiempo int NOT NULL,
FOREIGN KEY (id_unidad_venta) REFERENCES unidad_venta (id_unidad_venta)
);
create table plato(
id_plato int primary key auto_increment,
nombre VARCHAR(50) NOT NULL,
precio_de_venta double NOT NULL,
costo_de_plato double NOT NULL,
unidad_venta_id int NOT NULL,
FOREIGN KEY (unidad_venta_id) REFERENCES unidad_venta(id_unidad_venta)
);
create table pedido(
id_pedido int primary key AUTO_INCREMENT,
fecha_transaccion date not null,
unidad_venta_id int NOT NULL,
FOREIGN KEY (unidad_venta_id) REFERENCES unidad_venta(id_unidad_venta)
);
CREATE TABLE item_pedido(
  id_item_pedido INT PRIMARY KEY AUTO_INCREMENT,
  cantidad INT NOT NULL,
  plato_id INT NOT NULL,
  FOREIGN KEY (plato_id) REFERENCES plato(id_plato),
  pedido_id int not null,
  foreign key (pedido_id) references pedido(id_pedido)
);
